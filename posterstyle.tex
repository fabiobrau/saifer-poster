%% ------------------------------------------------------------
%%  Section box:  \begin{posterbox}[Title] ... \end{posterbox}
%% ------------------------------------------------------------
\newtcolorbox{posterbox}[1][]{%
  enhanced,
  colback=white,
  colframe=postercolor,
  boxrule=2pt,
  arc=0.25cm,
  rounded corners,
  fonttitle=\large\color{white},
  coltitle=white,
  title={#1},
  attach boxed title to top left={yshift=-0.75cm, xshift=1.5cm},
  boxed title style={
    colback=postercolor,
    rounded corners,
    arc=0.3cm,
    boxrule=0pt,
    top=10pt, bottom=10pt, left=14pt, right=14pt,
  },
  top=0.5cm,
  bottom=0.05cm,
  left=0.2cm,
  right=0.2cm,
  before upper={\vspace{5mm}},
  after upper={\vspace{5mm}},
  width=\linewidth,
}

%% ------------------------------------------------------------
%%  Header
%%  Fill in with \postertitle, \postersubtitle, \posterauthors,
%%  \posteraffiliations and \posterlogos (see main.tex),
%%  then call \makeposterheader after \begin{document}.
%% ------------------------------------------------------------
\newcommand{\@postertitle}{Poster Title}
\newcommand{\@postersubtitle}{}
\newcommand{\@posterauthors}{}
\newcommand{\@posteraffiliations}{}
\newcommand{\@posterlogos}{}
\newcommand{\postertitle}[1]{\renewcommand{\@postertitle}{#1}}
\newcommand{\postersubtitle}[1]{\renewcommand{\@postersubtitle}{#1}}
\newcommand{\posterauthors}[1]{\renewcommand{\@posterauthors}{#1}}
\newcommand{\posteraffiliations}[1]{\renewcommand{\@posteraffiliations}{#1}}
\newcommand{\posterlogos}[1]{\renewcommand{\@posterlogos}{#1}}

\newcommand{\makeposterheader}{%
  \noindent\hspace{\oddsidemargin}\colorbox{postercolor}{%
    \parbox{\paperwidth}{%
      \vspace{1.5cm}
      \centering
      \color{white}
      {\Huge\textsc{\@postertitle}}\\[1cm]
      \ifx\@postersubtitle\@empty\else{\Large \@postersubtitle}\\[1cm]\fi
      {\large\textit{\@posterauthors}}\\[0.3cm]
      \@posteraffiliations
      \vspace{1.5cm}
    }%
  }
  \ifx\@posterlogos\@empty\else
  \begin{minipage}[t]{\linewidth}
    \centering
    \vspace{1cm}
    \@posterlogos
  \end{minipage}\vspace{0.5cm}
  \fi
  \par
}

%% ------------------------------------------------------------
%%  Bibliography inside a box (no "References" heading)
%% ------------------------------------------------------------
\newcommand{\posterbibliography}[2]{%
  \bibliographystyle{#1}%
  \begingroup
  \renewcommand{\section}[2]{}%
  \bibliography{#2}%
  \endgroup
}

%% ------------------------------------------------------------
%%  Code listings
%% ------------------------------------------------------------
\lstdefinelanguage{PythonCustom}{
    language=Python,
    morekeywords=[2]{List,Dict,Optional,Tuple,Union,Any,Tensor,torch},
    morekeywords=[3]{zeros,ones,run},
    keywordstyle=\color{codeblue},
    keywordstyle=[2]\color{codesky},
    keywordstyle=[3]\color{codegreen},
}

\lstset{
    language=PythonCustom,
    backgroundcolor=\color{codebackground},
    basicstyle=\ttfamily\small\color{codetext},
    commentstyle=\color{codegray}\itshape,
    stringstyle=\color{codegreen},
    lineskip=2pt,
    xleftmargin=2pt,
    showstringspaces=false,
    breaklines=true,
    frame=single,
    framerule=0mm,
    rulecolor=\color{codegray},
    captionpos=b,
    escapeinside={\%*}{*\%},
}

%% ------------------------------------------------------------
%%  Generic logo placeholder:  \logoplaceholder[height]{Label}
%%  Replace with \includegraphics once you have the real logo.
%% ------------------------------------------------------------
\newcommand{\logoplaceholder}[2][0.08\textwidth]{%
  \tcbox[colback=postercolor!8, colframe=postercolor, boxrule=1.5pt,
         arc=0.2cm, left=0pt, right=0pt, top=0pt, bottom=0pt,
         nobeforeafter, box align=center]{%
    \parbox[c][#1][c]{1.9\dimexpr#1\relax}{\centering\normalsize\color{postercolor}\textbf{#2}}}%
}
